37template <typename T>
38void OpenadsDemoService::declareAndLoadParameter(const std::string& name,
39 T& param,
40 const std::string& description,
41 const bool add_to_auto_reconfigurable_params,
42 const bool is_required,
43 const bool read_only,
44 const std::optional<double>& from_value,
45 const std::optional<double>& to_value,
46 const std::optional<double>& step_value,
47 const std::string& additional_constraints) {
48 rcl_interfaces::msg::ParameterDescriptor param_desc;
49 param_desc.description = description;
50 param_desc.additional_constraints = additional_constraints;
51 param_desc.read_only = read_only;
52
53 auto type = rclcpp::ParameterValue(param).get_type();
54
55 if (from_value.has_value() && to_value.has_value()) {
56 if constexpr (std::is_integral_v<T>) {
57 rcl_interfaces::msg::IntegerRange range;
58 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
59 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
60 param_desc.integer_range = {range};
61 } else if constexpr (std::is_floating_point_v<T>) {
62 rcl_interfaces::msg::FloatingPointRange range;
63 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
64 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
65 param_desc.floating_point_range = {range};
66 } else {
67 RCLCPP_WARN(this->get_logger(), "Parameter type of parameter '%s' does not support specifying a range", name.c_str());
68 }
69 }
70
71 this->declare_parameter(name, type, param_desc);
72
73 try {
74 param = this->get_parameter(name).get_value<T>();
75 std::stringstream ss;
76 ss << "Loaded parameter '" << name << "': ";
77 if constexpr (is_vector_v<T>) {
78 ss << "[";
79 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "");
80 ss << "]";
81 } else {
82 ss << param;
83 }
84 RCLCPP_INFO_STREAM(this->get_logger(), ss.str());
85 } catch (rclcpp::exceptions::ParameterUninitializedException&) {
86 if (is_required) {
87 RCLCPP_FATAL_STREAM(this->get_logger(), "Missing required parameter '" << name << "', exiting");
88 exit(EXIT_FAILURE);
89 } else {
90 std::stringstream ss;
91 ss << "Missing parameter '" << name << "', using default value: ";
92 if constexpr (is_vector_v<T>) {
93 ss << "[";
94 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "");
95 ss << "]";
96 } else {
97 ss << param;
98 }
99 RCLCPP_WARN_STREAM(this->get_logger(), ss.str());
100 this->set_parameters({rclcpp::Parameter(name, rclcpp::ParameterValue(param))});
101 }
102 }
103
104 if (add_to_auto_reconfigurable_params) {
105 std::function<void(const rclcpp::Parameter&)> setter = [&param](const rclcpp::Parameter& p) { param = p.get_value<T>(); };
106 auto_reconfigurable_params_.push_back(std::make_tuple(name, setter));
107 }
108}

110rcl_interfaces::msg::SetParametersResult OpenadsDemoService::parametersCallback(
111 const std::vector<rclcpp::Parameter>& parameters) {
112 for (const auto& param : parameters) {
113 for (auto& auto_reconfigurable_param : auto_reconfigurable_params_) {
114 if (param.get_name() == std::get<0>(auto_reconfigurable_param)) {
115 std::get<1>(auto_reconfigurable_param)(param);
116 RCLCPP_INFO(this->get_logger(), "Reconfigured parameter '%s' to: %s", param.get_name().c_str(),
117 param.value_to_string().c_str());
118 break;
119 }
120 }
121 }
122
123 rcl_interfaces::msg::SetParametersResult result;
124 result.successful = true;
125
126 return result;
127}

129void OpenadsDemoService::setup() {
130 // callback for dynamic parameter configuration
131 parameters_callback_ =
132 this->add_on_set_parameters_callback(std::bind(&OpenadsDemoService::parametersCallback, this, std::placeholders::_1));
133
134 // subscriber for handling incoming messages
135 subscriber_ = this->create_subscription<perception_msgs::msg::EgoData>(
136 "~/input", 10, std::bind(&OpenadsDemoService::topicCallback, this, std::placeholders::_1));
137 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", subscriber_->get_topic_name());
138
139 // publisher for publishing outgoing messages
140 publisher_ = this->create_publisher<perception_msgs::msg::EgoData>("~/output", 10);
141 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", publisher_->get_topic_name());
142
143 // service server for handling service calls
144 service_server_ = this->create_service<std_srvs::srv::SetBool>(
145 "~/service", std::bind(&OpenadsDemoService::serviceCallback, this, std::placeholders::_1, std::placeholders::_2));
146
147 // action server for handling action goal requests
148 action_server_ = rclcpp_action::create_server<openads_demo_service_interfaces::action::Fibonacci>(
149 this, "~/action", std::bind(&OpenadsDemoService::actionHandleGoal, this, std::placeholders::_1, std::placeholders::_2),
150 std::bind(&OpenadsDemoService::actionHandleCancel, this, std::placeholders::_1),
151 std::bind(&OpenadsDemoService::actionHandleAccepted, this, std::placeholders::_1));
152
153 // timer for repeatedly invoking a callback
154 timer_ = this->create_wall_timer(std::chrono::duration<double>(1.0), std::bind(&OpenadsDemoService::timerCallback, this));
155
156 // setup diagnostic updater
157 diagnostic_updater_.setHardwareID("none");
158 diagnostic_updater_.add("Health", this, &OpenadsDemoService::health);
159
160 // add diagnostic task for monitoring topic subscription with a moving average over min. 5 incoming messages based on expected minimum frequency
161 const int topic_diagnostic_frequency_window_size =
162 std::ceil(5 / (diagnostic_updater_.getPeriod().seconds() * topic_diagnostic_config_.min_frequency));
163 topic_diagnostic_ = std::make_unique<diagnostic_updater::TopicDiagnostic>(
164 "~/input", diagnostic_updater_,
165 diagnostic_updater::FrequencyStatusParam(&topic_diagnostic_config_.min_frequency, &topic_diagnostic_config_.max_frequency,
166 0.0, topic_diagnostic_frequency_window_size),
167 diagnostic_updater::TimeStampStatusParam(topic_diagnostic_config_.min_acceptable_timestamp_delta,
168 topic_diagnostic_config_.max_acceptable_timestamp_delta));
169
170 // add diagnostic task for monitoring topic publisher with a moving average over min. 5 incoming messages based on expected minimum frequency
171 const int diagnosed_publisher_frequency_window_size =
172 std::ceil(5 / (diagnostic_updater_.getPeriod().seconds() * diagnosed_publisher_config_.min_frequency));
173 diagnosed_publisher_ = std::make_unique<diagnostic_updater::DiagnosedPublisher<perception_msgs::msg::EgoData>>(
174 publisher_, diagnostic_updater_,
175 diagnostic_updater::FrequencyStatusParam(&diagnosed_publisher_config_.min_frequency,
176 &diagnosed_publisher_config_.max_frequency, 0.0,
177 diagnosed_publisher_frequency_window_size),
178 diagnostic_updater::TimeStampStatusParam(diagnosed_publisher_config_.min_acceptable_timestamp_delta,
179 diagnosed_publisher_config_.max_acceptable_timestamp_delta));
180}

182void OpenadsDemoService::topicCallback(const perception_msgs::msg::EgoData::ConstSharedPtr& msg) {
183 topic_diagnostic_->tick(msg->header.stamp);
184 RCLCPP_INFO(this->get_logger(), "Message received with stamp: '%d'", msg->header.stamp.sec);
185 RCLCPP_INFO(this->get_logger(), "x=%f, y=%f, z=%f, yaw=%f", perception_msgs::object_access::getX(*msg),
186 perception_msgs::object_access::getY(*msg), perception_msgs::object_access::getZ(*msg),
187 perception_msgs::object_access::getYawInDeg(*msg));
188
189 // publish message
190 perception_msgs::msg::EgoData out_msg;
191 out_msg = *msg;
192 diagnosed_publisher_->publish(out_msg);
193 RCLCPP_INFO(this->get_logger(), "Message published with stamp: '%d'", out_msg.header.stamp.sec);
194}

230void OpenadsDemoService::actionExecute(
231 const std::shared_ptr<rclcpp_action::ServerGoalHandle<openads_demo_service_interfaces::action::Fibonacci>> goal_handle) {
232 RCLCPP_INFO(this->get_logger(), "Executing action goal");
233
234 // define a sleeping rate between computing individual Fibonacci numbers
235 rclcpp::Rate loop_rate(1);
236
237 // create handy accessors for the action goal, feedback, and result
238 const auto goal = goal_handle->get_goal();
239 auto feedback = std::make_shared<openads_demo_service_interfaces::action::Fibonacci::Feedback>();
240 auto result = std::make_shared<openads_demo_service_interfaces::action::Fibonacci::Result>();
241
242 // initialize the Fibonacci sequence
243 auto& partial_sequence = feedback->partial_sequence;
244 partial_sequence.push_back(0);
245 partial_sequence.push_back(1);
246
247 // compute the Fibonacci sequence up to the requested order n
248 for (int i = 1; i < goal->order && rclcpp::ok(); ++i) {
249 // cancel, if requested
250 if (goal_handle->is_canceling()) {
251 result->sequence = feedback->partial_sequence;
252 goal_handle->canceled(result);
253 RCLCPP_INFO(this->get_logger(), "Action goal canceled");
254 return;
255 }
256
257 // compute the next Fibonacci number
258 partial_sequence.push_back(partial_sequence[i] + partial_sequence[i - 1]);
259
260 // publish the current sequence as action feedback
261 goal_handle->publish_feedback(feedback);
262 RCLCPP_INFO(this->get_logger(), "Publishing action feedback");
263
264 // sleep before computing the next Fibonacci number
265 loop_rate.sleep();
266 }
267
268 // finish by publishing the action result
269 if (rclcpp::ok()) {
270 result->sequence = partial_sequence;
271 goal_handle->succeed(result);
272 RCLCPP_INFO(this->get_logger(), "Goal succeeded");
273 }
274}

276void OpenadsDemoService::timerCallback() {

279 // TODO(unknown): Remove this demonstration of health status and implement real health checks using `setHealth()`

292 health_.status = diagnostic_msgs::msg::DiagnosticStatus::STALE;

295 health_.status = diagnostic_msgs::msg::DiagnosticStatus::ERROR;
